cmake_minimum_required(VERSION 3.24)
project(blossom_installer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick)
add_executable(blossom-installer main.cpp)
target_link_libraries(blossom-installer PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick)
target_compile_options(blossom-installer PRIVATE -Wall -Wextra -Wpedantic -Werror)
