cmake_minimum_required(VERSION 3.24)
project(blossom_desktop_launcher LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core DBus)
add_executable(blossom-desktop-launcher main.cpp)
target_link_libraries(blossom-desktop-launcher PRIVATE Qt6::Core Qt6::DBus)
target_compile_options(blossom-desktop-launcher PRIVATE -Wall -Wextra -Wpedantic -Werror)
